const express = require('express'); const http = require('http'); const { Server } = require('socket.io'); // ⚠️ ضع اسمك أو رمزك الخاص هنا. من يكتب هذا الاسم في التطبيق يصبح المالك فوراً! const MY_SECRET_OWNER_NAME = "Abood_99"; const app = express(); const server = http.createServer(app); const io = new Server(server); let chatRooms = { "العراق": { "بغداد الثقافية": { owner: MY_SECRET_OWNER_NAME, admins: [], members: [] }, "جلسة شبابية": { owner: MY_SECRET_OWNER_NAME, admins: [], members: [] } }, "السعودية": { "مجلس الرياض": { owner: MY_SECRET_OWNER_NAME, admins: [], members: [] } }, "مصر": { "قهوة المحروسة": { owner: MY_SECRET_OWNER_NAME, admins: [], members: [] } } }; let bannedUsers = new Set(); let userRoles = {}; app.get('/', (req, res) => { res.send(` تطبيق دردشة المالك المتكامل

بوابة التحكم والدردشة

لوحة تحكم الإدارة

`); }); io.on('connection', (socket) => { socket.emit('initData', chatRooms); socket.on('joinRoom', ({ username, country, room }) => { if (bannedUsers.has(username)) { socket.emit('errorMessage', 'أنت محظور من دخول هذا التطبيق تماماً.'); return; } socket.join(room); socket.username = username; socket.room = room; socket.country = country; // التحقق من الاسم فقط لمنح رتبة المالك if (username === MY_SECRET_OWNER_NAME) { socket.role = "owner"; } else if (userRoles[username] === "admin") { socket.role = "admin"; } else { socket.role = "user"; } io.to(room).emit('sysMessage', `النظام: انضم ${username} إلى الغرفة برتبة (${socket.role === 'owner' ? 'المالك' : (socket.role === 'admin' ? 'مشرف' : 'عضو')})`); }); socket.on('chatMessage', (msg) => { if (bannedUsers.has(socket.username)) return; io.to(socket.room).emit('message', { user: socket.username, text: msg, role: socket.role }); }); socket.on('createRoom', ({ country, roomName }) => { if (socket.role === "owner") { if (!chatRooms[country]) chatRooms[country] = {}; chatRooms[country][roomName] = { owner: MY_SECRET_OWNER_NAME, admins: [], members: [] }; io.emit('initData', chatRooms); } else { socket.emit('errorMessage', 'خطأ: صلاحية إنشاء الغرف محصورة بمالك التطبيق فقط!'); } }); socket.on('adminAction', ({ action, targetUser }) => { if (socket.role === 'owner' || socket.role === 'admin') { if (action === 'ban') { if (targetUser === MY_SECRET_OWNER_NAME) return socket.emit('errorMessage', 'لا يمكن حظر المالك العام للتطبيق!'); bannedUsers.add(targetUser); io.emit('sysMessage', `تنبيه: تم حظر المستخدم [${targetUser}] نهائياً من الدردشة.`); } else if (action === 'promote') { userRoles[targetUser] = "admin"; io.emit('sysMessage', `مبروك: تم ترقية المستخدم [${targetUser}] إلى رتبة مشرف بنجاح.`); } } else { socket.emit('errorMessage', 'ليس لديك صلاحيات كافية لتنفيذ هذا الإجراء.'); } }); socket.on('disconnect', () => { if (socket.room) { io.to(socket.room).emit('sysMessage', `غادر ${socket.username} الدردشة.`); } }); }); const PORT = 3000; server.listen(PORT, () => console.log(`التطبيق يعمل على: http://localhost:${PORT}`));
